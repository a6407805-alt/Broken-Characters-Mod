// ShopAuditPopupManager.cs  -  drop into DealerMod/ next to DealerPlugin.cs
//
// Stress-to-Popup feedback loop for Max the Dealer.
//   * ShopStress.CurrentStress (0-100) drives how often ERO / appointment windows spawn.
//   * Open windows add stress every second, finishing them removes it (compounding loop).
//   * 100% stress = full-screen "Management Audit" lockdown + continuous barrage until the backlog is cleared.
// All UI lives on one ScreenSpaceOverlay canvas (sortingOrder 999, scaled from 1920x1080).
// Clicks are hit-tested by this file (no EventSystem needed), the cursor is freed with MPEventSystem.cursorOpenerCount.

using System;
using System.Collections.Generic;
using System.Reflection;
using BepInEx;
using BepInEx.Configuration;
using RoR2;
using UnityEngine;
using UnityEngine.UI;
using Object = UnityEngine.Object;
using Random = UnityEngine.Random;

namespace DealerMod
{
    // ------------------------------------------------------------------ stress

    /// <summary>Single source of truth for shop stress, 0-100. Your skills can call ShopStress.Add(...).</summary>
    public static class ShopStress
    {
        private static float _current;

        public static float CurrentStress
        {
            get { return _current; }
            set { _current = Mathf.Clamp(value, 0f, 100f); }
        }

        public static void Add(float amount) { CurrentStress = _current + amount; }
        public static void Reduce(float amount) { CurrentStress = _current - amount; }
    }

    public enum StressTier { Low, Medium, High, Max }

    // ------------------------------------------------------------------ manager

    public class ShopAuditPopupManager : MonoBehaviour
    {
        public const string DealerNameToken = "AHREN_DEALER_NAME";
        public static ShopAuditPopupManager Instance;

        // ---- config (rename a key to _v2 if you change its default later)
        public static ConfigEntry<bool> Enabled, TestWithAnySurvivor, BuiltInStress;
        public static ConfigEntry<KeyCode> CursorKey;
        public static ConfigEntry<float> MediumFrom, HighFrom;
        public static ConfigEntry<float> LowInterval, MedMin, MedMax, HighMin, HighMax, LockInterval;
        public static ConfigEntry<int> LowCap, MedCap, HighCap, HighBurst, AuditBacklog;
        public static ConfigEntry<float> AppointmentChance, EroRelief, ApptRelief;
        public static ConfigEntry<float> ClosePenalty, WrongPenalty, MismatchPenalty, OverstepPenalty;
        public static ConfigEntry<float> StressCreep, StressPerWindow, HitStressScale, AuditEndStress;
        public static ConfigEntry<float> SliderSensitivity, SliderTolerance;

        {

            TestWithAnySurvivor = c.Bind("Popups", "TestWithAnySurvivor", false, "Run the popups for any survivor (handy for testing).");
            CursorKey = c.Bind("Popups", "CursorKey", KeyCode.LeftAlt, "Hold this to free the mouse and click popups. During an audit the mouse is always free.");

            BuiltInStress = c.Bind("Stress", "BuiltInStress", true, "Use the built-in stress source (creep + damage taken). Turn off if your own code sets ShopStress.CurrentStress.");
            StressCreep = c.Bind("Stress", "CreepPerSecond", 0.5f, "Built-in stress gained per second.");
            HitStressScale = c.Bind("Stress", "DamageStressScale", 0.6f, "Built-in stress per % of max health lost (0.6 = losing 10% health adds 6 stress).");
            StressPerWindow = c.Bind("Stress", "StressPerOpenWindowPerSecond", 0.25f, "Stress each open ticket adds per second (this is what makes the loop compound).");
            MediumFrom = c.Bind("Stress", "MediumTierFrom", 41f, "Stress where popups get frequent.");
            HighFrom = c.Bind("Stress", "HighTierFrom", 75f, "Stress where the malware-style barrage starts.");
            AuditEndStress = c.Bind("Stress", "StressAfterAudit", 40f, "Stress you are reset to when an audit is cleared.");

            LowInterval = c.Bind("Spawn - Low (0-40%)", "SecondsBetween", 24f, "Average seconds between popups.");
            LowCap = c.Bind("Spawn - Low (0-40%)", "MaxOpen", 2, "Max tickets open at once.");

            MedMax = c.Bind("Spawn - Medium (41-74%)", "SecondsMax", 12f, "Longest gap.");

            HighMin = c.Bind("Spawn - High (75-99%)", "SecondsMin", 1.5f, "Shortest gap between bursts.");
            HighMax = c.Bind("Spawn - High (75-99%)", "SecondsMax", 3f, "Longest gap between bursts.");
            HighBurst = c.Bind("Spawn - High (75-99%)", "MaxWindowsPerBurst", 3, "Each burst spawns 1 to this many windows.");
            HighCap = c.Bind("Spawn - High (75-99%)", "MaxOpen", 10, "Max tickets open at once.");
            LockInterval = c.Bind("Spawn - Audit (100%)", "SecondsBetween", 0.8f, "Gap between windows during the audit barrage.");
            AuditBacklog = c.Bind("Spawn - Audit (100%)", "BacklogToClear", 8, "Tickets you must finish to end the audit.");

            AppointmentChance = c.Bind("Tickets", "AppointmentChance", 0.45f, "Chance a popup is an appointment window instead of an ERO.");
            EroRelief = c.Bind("Tickets", "EroStressRelief", 35f, "Stress removed by a completed repair order.");
            ApptRelief = c.Bind("Tickets", "AppointmentStressRelief", 10f, "Stress removed by a booked appointment.");
            ClosePenalty = c.Bind("Tickets", "CloseWithoutFinishingStress", 10f, "Stress added if you X out a ticket.");
            WrongPenalty = c.Bind("Tickets", "WrongStampStress", 3f, "Stress added by a bad LARGE COMPLETED stamp.");
            MismatchPenalty = c.Bind("Tickets", "DateMismatchStress", 2f, "Stress added by a date mismatch.");
            OverstepPenalty = c.Bind("Tickets", "PromiseOvershootStress", 3f, "Stress added when the promise time slider overshoots.");
            SliderSensitivity = c.Bind("Tickets", "SliderSensitivity", 3f, "Promise time slider speed multiplier (higher = twitchier).");
            SliderTolerance = c.Bind("Tickets", "SliderToleranceMinutes", 10f, "How close the promise time must be to the target.");
        }

        public static void Install(BaseUnityPlugin plugin)
        {
            BindConfig(plugin.Config);
            if (Instance == null) Instance = plugin.gameObject.AddComponent<ShopAuditPopupManager>();
        }

        // ---- state
        private CharacterBody _body;
        private float _lastHp = -1f;
        private Canvas _cv;
        private readonly List<PopupWindow> _windows = new List<PopupWindow>();
        private float _spawnTimer = 3f;
        private float _postAuditCooldown;
        private bool _locked;
        private int _backlog;
        private bool _cursorHeld;
        private Vector2 _lastMouse;
        private Ctl _dragCtl;
        private PopupWindow _dragWindow;

        // static layers
        private GameObject _auditRoot;
        private Image _auditDim, _hudFill;
        private Text _auditSub, _hudText, _hint, _toast;
        private float _toastTimer;

        public Canvas Cv { get { return _cv; } }
        public bool Locked { get { return _locked; } }

        public StressTier CurrentTier
        {
            get { return _locked ? StressTier.Max : TierFor(ShopStress.CurrentStress); }
        }

        public static StressTier TierFor(float stress)
        {
            int p = Mathf.FloorToInt(stress);
            if (p >= 100) return StressTier.Max;
            if (p >= Mathf.RoundToInt(HighFrom.Value)) return StressTier.High;
            if (p >= Mathf.RoundToInt(MediumFrom.Value)) return StressTier.Medium;
            return StressTier.Low;
        }

        public static void AddStress(float amount) { ShopStress.Add(amount); }

        public static void Sfx(string evt)
        {
            if (Instance == null || Instance._body == null) return;
            try { Util.PlaySound(evt, Instance._body.gameObject); } catch (Exception) { }
        }

        // ---------------------------------------------------------------- frame

        private void Update()
        {
            CharacterBody body = FindBody();
            if (body == null)
            {
                if (_body != null || _windows.Count > 0 || _locked) ResetAll();
                _body = null;
                return;
            }

            _body = body;
            float dt = Time.deltaTime;
            EnsureCanvas();

            SimulateStress(body, dt);
            UpdateLockdown();
            UpdateSpawning(dt);
            UpdateCursor();
            HandleMouse();

            foreach (PopupWindow w in new List<PopupWindow>(_windows))
            {
                if (w.Root) w.Tick(dt);
            }

            RefreshHud(dt);
        }

        private void Awake()
        {
            Run.onRunStartGlobal += OnRunStart;
        }

        private void OnRunStart(Run run)
        {
            ShopStress.CurrentStress = 0f;
            ResetAll();
        }

        private void OnDestroy()
        {
            Run.onRunStartGlobal -= OnRunStart;
            SetCursor(false);
        }

        private CharacterBody FindBody()
        {
            if (Enabled == null || !Enabled.Value) return null;
            LocalUser lu = LocalUserManager.GetFirstLocalUser();
            if (lu == null) return null;
            CharacterBody b = lu.cachedBody;
            if (!b || !b.healthComponent || !b.healthComponent.alive) return null;
            if (!TestWithAnySurvivor.Value && b.baseNameToken != DealerNameToken) return null;
            return b;
        }

        private void SimulateStress(CharacterBody body, float dt)
        {
            if (BuiltInStress.Value)
            {
                ShopStress.Add(StressCreep.Value * dt);
                float hp = body.healthComponent.health;
                if (_lastHp >= 0f && hp < _lastHp && body.maxHealth > 0f)
                    ShopStress.Add((_lastHp - hp) / body.maxHealth * 100f * HitStressScale.Value);
                _lastHp = hp;
            }

            // The compounding part: every open ticket keeps adding stress.
            ShopStress.Add(CountTickets() * StressPerWindow.Value * dt);
        }

        private void UpdateLockdown()
        {
            if (!_locked && ShopStress.CurrentStress >= 100f) StartAudit();
            if (_locked && _backlog <= 0) EndAudit();
        }

        private void StartAudit()
        {
            _locked = true;
            _backlog = Mathf.Max(1, AuditBacklog.Value);
            _spawnTimer = 0.3f;
            ShowToast("MANAGEMENT AUDIT", 2.2f);
            Sfx("Play_UI_menuCancel");
        }

        private void EndAudit()
        {
            _locked = false;
            _backlog = 0;
            ShopStress.CurrentStress = AuditEndStress.Value;
            _postAuditCooldown = 6f;
            ShowToast("AUDIT PASSED", 2.5f);
        }

        private void UpdateSpawning(float dt)
        {
            if (_postAuditCooldown > 0f) { _postAuditCooldown -= dt; return; }

            _spawnTimer -= dt;
            if (_spawnTimer > 0f) return;

            int cap;
            int burst = 1;
            float gap;
            switch (CurrentTier)
            {
                case StressTier.Low:
                    cap = LowCap.Value;
                    gap = LowInterval.Value * Random.Range(0.75f, 1.25f);
                    break;
                case StressTier.Medium:
                    cap = MedCap.Value;
                    gap = Random.Range(MedMin.Value, Mathf.Max(MedMin.Value, MedMax.Value));
                    break;
                case StressTier.High:
                    cap = HighCap.Value;
                    burst = Random.Range(1, Mathf.Max(1, HighBurst.Value) + 1);
                    gap = Random.Range(HighMin.Value, Mathf.Max(HighMin.Value, HighMax.Value));
                    break;
                default:
                    cap = Mathf.Max(HighCap.Value, 14);
                    gap = LockInterval.Value;
                    break;
            }

            _spawnTimer = Mathf.Max(0.2f, gap);

            for (int i = 0; i < burst; i++)
            {
                int open = CountTickets();
                if (open >= cap) break;
                if (_locked && open >= _backlog) break; // never spawn more than you can clear
                SpawnRandom();
            }
        }

        private int CountTickets()
        {
            int n = 0;
            for (int i = 0; i < _windows.Count; i++)
                if (!_windows[i].IsError && !_windows[i].Finished) n++;
            return n;
        }

        private void SpawnRandom()
        {
            if (Random.value < AppointmentChance.Value) new AppointmentWindow(this);
            else new EroWindow(this);
        }

        public void SpawnError(string code, string text)
        {
            new ErrorWindow(this, code, text);
        }

        // ---------------------------------------------------------------- windows

        public void Register(PopupWindow w)
        {
            _windows.Add(w);
            w.Root.SetAsLastSibling();
            Sfx("Play_UI_menuClick");
        }

        public void BringToFront(PopupWindow w)
        {
            if (!w.Root) return;
            _windows.Remove(w);
            _windows.Add(w);
            w.Root.SetAsLastSibling();
        }

        public void CloseWindow(PopupWindow w)
        {
            _windows.Remove(w);
            if (_dragWindow == w) _dragWindow = null;
            if (w.Root) Object.Destroy(w.Root.gameObject);
        }

        public void AbandonWindow(PopupWindow w)
        {
            if (_locked) { Sfx("Play_UI_menuCancel"); return; }
            AddStress(ClosePenalty.Value);
            CloseWindow(w);
        }

        public void CompleteTicket(float relief)
        {
            ShopStress.Reduce(relief);
            if (_locked) _backlog = Mathf.Max(0, _backlog - 1);
        }

        private void ClearWindows()
        {
            foreach (PopupWindow w in new List<PopupWindow>(_windows))
                if (w.Root) Object.Destroy(w.Root.gameObject);
            _windows.Clear();
            _dragCtl = null;
            _dragWindow = null;
        }

        private void ResetAll()
        {
            if (_locked) ShopStress.CurrentStress = Mathf.Min(ShopStress.CurrentStress, AuditEndStress.Value);
            ClearWindows();
            _locked = false;
            _backlog = 0;
            _lastHp = -1f;
            _spawnTimer = 3f;
            SetCursor(false);
            if (_cv) _cv.gameObject.SetActive(false);
        }

        public Vector2 CanvasSize()
        {
            RectTransform rt = _cv ? _cv.transform as RectTransform : null;
            if (rt != null && rt.rect.width > 100f) return rt.rect.size;
            return new Vector2(1920f, 1080f);
        }

        public Vector2 PickPosition(float w, float h)
        {
            Vector2 cs = CanvasSize();
            float maxX = Mathf.Max(40f, cs.x - w - 40f);
            float maxY = Mathf.Max(80f, cs.y - h - 30f);
            Vector2 p = new Vector2(Random.Range(30f, maxX), Random.Range(70f, maxY));

            // At medium stress and up, windows like to land on top of each other.
            if (_windows.Count > 0 && CurrentTier >= StressTier.Medium && Random.value < 0.35f)
            {
                PopupWindow last = _windows[_windows.Count - 1];
                if (last.Root)
                {
                    Vector2 a = last.Root.anchoredPosition;
                    p = new Vector2(a.x + Random.Range(-90f, 90f), -a.y + Random.Range(-70f, 70f));
                    p.x = Mathf.Clamp(p.x, 10f, maxX + 20f);
                    p.y = Mathf.Clamp(p.y, 50f, maxY + 20f);
                }
            }
            return p;
        }

        // ---------------------------------------------------------------- input

        private void UpdateCursor()
        {
            bool want = _locked || (_windows.Count > 0 && Input.GetKey(CursorKey.Value));
            SetCursor(want);
        }

        private void SetCursor(bool want)
        {
            if (want == _cursorHeld) return;
            _cursorHeld = want;
            AdjustCursorOpeners(want ? 1 : -1);
            if (!want) { _dragCtl = null; _dragWindow = null; }
        }

        private static object GetEventSystem()
        {
            try
            {
                LocalUser lu = LocalUserManager.GetFirstLocalUser();
                if (lu == null) return null;
                BindingFlags f = BindingFlags.Instance | BindingFlags.Public | BindingFlags.NonPublic;
                PropertyInfo p = typeof(LocalUser).GetProperty("eventSystem", f);
                if (p != null) return p.GetValue(lu, null);
                FieldInfo fi = typeof(LocalUser).GetField("eventSystem", f);
                if (fi != null) return fi.GetValue(lu);
            }
            catch (Exception) { }
            return null;
        }

        private static void AdjustCursorOpeners(int delta)
        {
            try
            {
                object es = GetEventSystem();
                if (es == null) return;
                BindingFlags f = BindingFlags.Instance | BindingFlags.Public | BindingFlags.NonPublic;
                Type t = es.GetType();
                FieldInfo fi = t.GetField("cursorOpenerCount", f);
                if (fi != null)
                {
                    fi.SetValue(es, Mathf.Max(0, (int)fi.GetValue(es) + delta));
                    return;
                }
                PropertyInfo pi = t.GetProperty("cursorOpenerCount", f);
                if (pi != null && pi.CanRead && pi.CanWrite)
                    pi.SetValue(es, Mathf.Max(0, (int)pi.GetValue(es, null) + delta), null);
            }
            catch (Exception) { }
        }

        private void HandleMouse()
        {
            Vector2 mp = Input.mousePosition;
            if (!_cursorHeld) { _lastMouse = mp; _dragCtl = null; _dragWindow = null; return; }

            if (Input.GetMouseButtonDown(0)) OnPress(mp);

            if (Input.GetMouseButton(0))
            {
                float scale = _cv ? Mathf.Max(0.01f, _cv.scaleFactor) : 1f;
                Vector2 delta = (mp - _lastMouse) / scale;
                if (_dragCtl != null && _dragCtl.Rt) _dragCtl.Drag(delta);
                else if (_dragWindow != null && _dragWindow.Root) MoveWindow(_dragWindow, delta);
            }

            if (Input.GetMouseButtonUp(0))
            {
                if (_dragCtl != null && _dragCtl.Rt) _dragCtl.Release();
                _dragCtl = null;
                _dragWindow = null;
            }

            _lastMouse = mp;
        }

        private void OnPress(Vector2 sp)
        {
            _dragCtl = null;
            _dragWindow = null;
            for (int i = _windows.Count - 1; i >= 0; i--)
            {
                PopupWindow w = _windows[i];
                if (!w.Root || !RectTransformUtility.RectangleContainsScreenPoint(w.Root, sp, null)) continue;

                BringToFront(w);
                for (int c = w.Ctls.Count - 1; c >= 0; c--)
                {
                    Ctl ctl = w.Ctls[c];
                    if (!ctl.Hit(sp)) continue;
                    _dragCtl = ctl;
                    ctl.Press(sp);
                    return;
                }

                if (w.TitleBar && RectTransformUtility.RectangleContainsScreenPoint(w.TitleBar, sp, null))
                    _dragWindow = w;
                return;
            }
        }

        private void MoveWindow(PopupWindow w, Vector2 delta)
        {
            Vector2 cs = CanvasSize();
            Vector2 p = w.Root.anchoredPosition + delta;
            p.x = Mathf.Clamp(p.x, -w.Width + 80f, cs.x - 80f);
            p.y = Mathf.Clamp(p.y, -(cs.y - 40f), 0f);
            w.Root.anchoredPosition = p;
        }

        // ---------------------------------------------------------------- canvas + HUD

        private void EnsureCanvas()
        {
            if (_cv != null)
            {
                if (!_cv.gameObject.activeSelf) _cv.gameObject.SetActive(true);
                return;
            }

            GameObject go = new GameObject("DealerAuditCanvas");
            Object.DontDestroyOnLoad(go);
            _cv = go.AddComponent<Canvas>();
            _cv.renderMode = RenderMode.ScreenSpaceOverlay;
            _cv.sortingOrder = 999;

            CanvasScaler cs = go.AddComponent<CanvasScaler>();
            cs.uiScaleMode = CanvasScaler.ScaleMode.ScaleWithScreenSize;
            cs.referenceResolution = new Vector2(1920f, 1080f);
            cs.screenMatchMode = CanvasScaler.ScreenMatchMode.MatchWidthOrHeight;
            cs.matchWidthOrHeight = 0.5f;

            BuildStaticLayers();
        }

        private void BuildStaticLayers()
        {
            // Audit overlay (always the bottom layer so windows sit on top of it)
            RectTransform ar = new GameObject("AuditOverlay", typeof(RectTransform)).GetComponent<RectTransform>();
            ar.SetParent(_cv.transform, false);
            ar.anchorMin = Vector2.zero;
            ar.anchorMax = Vector2.one;
            ar.offsetMin = Vector2.zero;
            ar.offsetMax = Vector2.zero;
            _auditDim = ar.gameObject.AddComponent<Image>();
            _auditDim.color = new Color(0.14f, 0f, 0f, 0.78f);
            _auditDim.raycastTarget = false;
            Ui.CenterLabel(ar, "MANAGEMENT AUDIT", 110f, 1600f, 140f, 100, new Color(1f, 0.2f, 0.2f), FontStyle.Bold);
            _auditSub = Ui.CenterLabel(ar, "", -10f, 1500f, 140f, 30, Color.white, FontStyle.Normal);
            _auditRoot = ar.gameObject;
            _auditRoot.SetActive(false);

            // Stress bar, top centre
            RectTransform hud = Ui.Anchored("StressHud", _cv.transform, new Vector2(0.5f, 1f), new Vector2(0.5f, 1f), new Vector2(0f, -14f), new Vector2(440f, 30f));
            Ui.Box(hud, 0f, 0f, 440f, 30f, new Color(0.04f, 0.04f, 0.06f, 0.85f), "HudBack");
            _hudFill = Ui.Box(hud, 2f, 2f, 0f, 26f, Color.green, "HudFill");
            _hudText = Ui.Label(hud, "", 0f, 0f, 440f, 30f, 15, Color.white, TextAnchor.MiddleCenter, FontStyle.Bold);

            // Hint, bottom centre
            RectTransform hr = Ui.Anchored("Hint", _cv.transform, new Vector2(0.5f, 0f), new Vector2(0.5f, 0f), new Vector2(0f, 40f), new Vector2(900f, 34f));
            _hint = Ui.Label(hr, "", 0f, 0f, 900f, 34f, 20, new Color(1f, 0.95f, 0.5f), TextAnchor.MiddleCenter, FontStyle.Bold);

            // Toast, above centre
            _toast = Ui.CenterLabel(_cv.transform, "", 280f, 1500f, 100f, 64, new Color(1f, 0.85f, 0.3f), FontStyle.Bold);
        }

        private void ShowToast(string text, float seconds)
        {
            if (!_toast) return;
            _toast.text = text;
            _toastTimer = seconds;
        }

        private static string TierName(StressTier t)
        {
            switch (t)
            {
                case StressTier.Low: return "CALM";
                case StressTier.Medium: return "BUSY";
                case StressTier.High: return "OVERLOADED";
                default: return "AUDIT";
            }
        }

        private static Color TierColor(StressTier t)
        {
            switch (t)
            {
                case StressTier.Low: return new Color(0.3f, 0.85f, 0.4f);
                case StressTier.Medium: return new Color(1f, 0.85f, 0.2f);
                case StressTier.High: return new Color(1f, 0.5f, 0.15f);
                default: return new Color(1f, 0.15f, 0.15f);
            }
        }

        private void RefreshHud(float dt)
        {
            float s = ShopStress.CurrentStress;
            StressTier t = CurrentTier;
            _hudFill.rectTransform.sizeDelta = new Vector2(436f * s / 100f, 26f);
            _hudFill.color = TierColor(t);
            _hudText.text = "SHOP STRESS " + Mathf.RoundToInt(s) + "%   -   " + TierName(t);

            _auditRoot.SetActive(_locked);
            if (_locked)
            {
                _auditDim.color = new Color(0.14f, 0f, 0f, 0.72f + 0.12f * Mathf.Sin(Time.unscaledTime * 6f));
                _auditSub.text = "ALL REPAIR ORDERS MUST BE RECONCILED BEFORE THE SHIFT RESUMES.\nBACKLOG: " + _backlog + " REMAINING";
            }

            if (!_locked && _windows.Count > 0 && !_cursorHeld)
                _hint.text = "HOLD [" + CursorKey.Value + "] AND CLICK TO WORK THE POPUPS";
            else
                _hint.text = "";

            if (_toastTimer > 0f)
            {
                _toastTimer -= dt;
                if (_toastTimer <= 0f) _toast.text = "";
            }
        }
    }

    // ------------------------------------------------------------------ ui helpers

    internal static class Ui
    {
        private static Font _font;

        public static Font GetFont()
        {
            if (_font) return _font;
            try { _font = Resources.GetBuiltinResource<Font>("Arial.ttf"); } catch (Exception) { }
            if (!_font)
            {
                try { _font = Resources.GetBuiltinResource<Font>("LegacyRuntime.ttf"); } catch (Exception) { }
            }
            if (!_font) _font = Font.CreateDynamicFontFromOSFont("Arial", 16);
            return _font;
        }

        public static RectTransform Anchored(string name, Transform parent, Vector2 anchor, Vector2 pivot, Vector2 pos, Vector2 size)
        {
            GameObject go = new GameObject(name, typeof(RectTransform));
            RectTransform rt = (RectTransform)go.transform;
            rt.SetParent(parent, false);
            rt.anchorMin = anchor;
            rt.anchorMax = anchor;
            rt.pivot = pivot;
            rt.anchoredPosition = pos;
            rt.sizeDelta = size;
            return rt;
        }

        /// <summary>Top-left anchored rect; x and y are measured from the parent's top-left, y downward.</summary>
        public static RectTransform Rect(string name, Transform parent, float x, float y, float w, float h)
        {
            return Anchored(name, parent, new Vector2(0f, 1f), new Vector2(0f, 1f), new Vector2(x, -y), new Vector2(w, h));
        }

        public static Image Box(Transform parent, float x, float y, float w, float h, Color c, string name)
        {
            RectTransform rt = Rect(name, parent, x, y, w, h);
            Image img = rt.gameObject.AddComponent<Image>();
            img.color = c;
            img.raycastTarget = false;

        public static void Shuffle<T>(IList<T> list)
        {
            for (int i = list.Count - 1; i > 0; i--)
            {
                int j = Random.Range(0, i + 1);
                T tmp = list[i];
                list[i] = list[j];
                list[j] = tmp;
            }
        }

    public abstract class Ctl
    {
        public RectTransform Rt;

        public bool Hit(Vector2 screenPoint)
        {
            return Rt && Rt.gameObject.activeInHierarchy && RectTransformUtility.RectangleContainsScreenPoint(Rt, screenPoint, null);
        }

        public virtual void Press(Vector2 screenPoint) { }
        public virtual void Drag(Vector2 deltaRef) { }
        public virtual void Release() { }
    }

    public class ButtonCtl : Ctl
    {
        public Action OnClick;
        public bool Enabled = true;
        public Image Bg;
        public Text Label;
        public Color Normal;
        public Color DisabledColor = new Color(0.58f, 0.58f, 0.58f, 1f);

        public override void Press(Vector2 screenPoint)
        {
            if (!Enabled) { ShopAuditPopupManager.Sfx("Play_UI_menuCancel"); return; }
            ShopAuditPopupManager.Sfx("Play_UI_menuClick");
            if (OnClick != null) OnClick();
        }

        public void SetEnabled(bool on)
        {
            Enabled = on;
            Refresh();
        }

        public void Refresh()
        {
            if (Bg) Bg.color = Enabled ? Normal : DisabledColor;
        }
    }

    public class CheckCtl : Ctl
    {
        public bool Checked;
        public Text Mark;
        public Action Changed;

        public override void Press(Vector2 screenPoint)
        {
            Checked = !Checked;
            if (Mark) Mark.text = Checked ? "X" : "";
            ShopAuditPopupManager.Sfx("Play_UI_menuClick");
            if (Changed != null) Changed();
        }
    }

    public class SliderCtl : Ctl
    {
        public float Value;
        public float TrackX;
        public float TrackWidth = 100f;
        public RectTransform Handle;
        public Action<float> Changed;

        public override void Drag(Vector2 deltaRef)
        {
            SetValue(Value + deltaRef.x / TrackWidth * ShopAuditPopupManager.SliderSensitivity.Value, true);
        }

        public void SetValue(float v, bool notify)
        {
            Value = Mathf.Clamp01(v);
            if (Handle)
                Handle.anchoredPosition = new Vector2(TrackX + Value * TrackWidth - Handle.sizeDelta.x * 0.5f, Handle.anchoredPosition.y);
            if (notify && Changed != null) Changed(Value);
        }
    }

            if (_close.Enabled != canClose) _close.SetEnabled(canClose);
        }

        private static char[] ScrambledLayout()
        {
            char[] layout = "QWERTYUIOPASDFGHJKLZXCVBNM".ToCharArray();
            List<int> slots = new List<int>();
            List<char> vowels = new List<char>();
            for (int i = 0; i < layout.Length; i++)
            {
                if ("AEIOU".IndexOf(layout[i]) >= 0) { slots.Add(i); vowels.Add(layout[i]); }
            }
            for (int tries = 0; tries < 20; tries++)
            {
                PopupData.Shuffle(vowels);
                int moved = 0;
                for (int k = 0; k < slots.Count; k++) if (layout[slots[k]] != vowels[k]) moved++;
                if (moved >= 4) break;
            }
            for (int k = 0; k < slots.Count; k++) layout[slots[k]] = vowels[k];
            return layout;
        }

            _slider = new SliderCtl { Rt = hit, Handle = handle, TrackX = tx, TrackWidth = tw };
            _slider.Changed = OnSlider;
            Ctls.Add(_slider);

            Ui.Label(g, "Click and drag the handle. It is extremely sensitive. Go past the target and the scheduler will complain.", 16f, 244f, Width - 32f, 40f, 13, new Color(0.3f, 0.3f, 0.3f, 1f));
            AddButton(g, Width * 0.5f - 110f, Height - 98f, 220f, 40f, "SET PROMISE TIME", 16, Green, Color.white, SetTime);
        }

        private void OnSlider(float v)
        {
            int m = MinutesFor(v);
            _timeText.text = FormatTime(m);
            if (m > _targetMin + 20 && _errCool <= 0f)
            {
                _errCool = 1.5f;
                _slider.SetValue(0f, false);
                _timeText.text = FormatTime(MinutesFor(0f));
                ShopAuditPopupManager.AddStress(ShopAuditPopupManager.OverstepPenalty.Value);
                Mgr.SpawnError("ERROR 0x8007000E", "PROMISE TIME EXCEEDS BAY CAPACITY.\nSELECT AN EARLIER TIME.");
            }
        }

        private void SetTime()
        {
            int m = MinutesFor(_slider.Value);
            int tol = Mathf.Max(5, Mathf.RoundToInt(ShopAuditPopupManager.SliderTolerance.Value));
            if (Mathf.Abs(m - _targetMin) <= tol) GoStep(4);
            else Penalize(ShopAuditPopupManager.MismatchPenalty.Value, "PROMISE TIME NOT ACCEPTABLE. TARGET IS " + FormatTime(_targetMin) + ".");
        }

        private void BuildProgressStep()
        {
            Transform g = _g[4];
            Ui.Label(g, "Posting appointment to the dealer management system...", 16f, 70f, Width - 32f, 24f, 14, Ink);
            Ui.Box(g, 59f, 119f, Width - 118f, 32f, new Color(0.2f, 0.2f, 0.2f, 1f), "BarBorder");
            Ui.Box(g, 60f, 120f, Width - 120f, 30f, new Color(0.45f, 0.45f, 0.45f, 1f), "BarBack");
            _fill = Ui.Box(g, 62f, 122f, 0f, 26f, new Color(0.1f, 0.6f, 0.2f, 1f), "BarFill");
            _pct = Ui.Label(g, "0%", 60f, 120f, Width - 120f, 30f, 16, Color.white, TextAnchor.MiddleCenter, FontStyle.Bold);
            _status = Ui.Label(g, "", 60f, 160f, Width - 120f, 26f, 14, Ink);

            _checkGroup = Ui.Rect("Checks", g, 0f, 0f, Width, Height);
            string[] labels =
            {
                "I certify the Promise Time is acceptable to the customer.",
                "I acknowledge all dates are subject to change without notice.",
                "I have read the Terms of Service (page 1 of 47)."
            };
            for (int i = 0; i < 3; i++)
            {
                float y = 205f + i * 40f;
                RectTransform hit = Ui.Rect("CheckHit", _checkGroup, 60f, y, Width - 120f, 34f);
                Ui.Box(_checkGroup, 60f, y + 5f, 24f, 24f, new Color(0.2f, 0.2f, 0.2f, 1f), "CheckBorder");
                Ui.Box(_checkGroup, 61f, y + 6f, 22f, 22f, Color.white, "CheckBox");
                Text mark = Ui.Label(_checkGroup, "", 61f, y + 6f, 22f, 22f, 18, Ink, TextAnchor.MiddleCenter, FontStyle.Bold);
                Ui.Label(_checkGroup, labels[i], 94f, y, Width - 154f, 34f, 14, Ink, TextAnchor.MiddleLeft);
                CheckCtl c = new CheckCtl { Rt = hit, Mark = mark };
                _checks[i] = c;
                Ctls.Add(c);
            }
            _checkGroup.gameObject.SetActive(false);
        }

        private bool AllChecked()
        {
            for (int i = 0; i < 3; i++) if (!_checks[i].Checked) return false;
            return true;
        }

        public override void Tick(float dt)
        {
            base.Tick(dt);
            if (_errCool > 0f) _errCool -= dt;

            if (_closing)
            {
                _closeT -= dt;
                if (_closeT <= 0f) Mgr.CloseWindow(this);
                return;
            }
            if (_step != 4) return;

            if (!_released)
            {
                _progT += dt;
                float t = Mathf.Clamp01(_progT / 10f);
                _prog = 0.99f * t;

                if (t >= 1f)
                {
                    _checkGroup.gameObject.SetActive(true);
                    _status.text = "COMPLIANCE VERIFICATION REQUIRED. THE PROGRESS BAR IS WAITING ON YOU.";
                    if (AllChecked()) _released = true;
                }
                else if (t < 0.25f) _status.text = "Contacting scheduling server...";
                else if (t < 0.5f) _status.text = "Validating customer record...";
                else if (t < 0.75f) _status.text = "Reserving service bay...";
                else _status.text = "Finalizing...";
            }
            else
            {
                _prog = Mathf.MoveTowards(_prog, 1f, dt * 1.6f);
                if (_prog >= 1f && !Finished) FinishBooking();
            }

            _fill.rectTransform.sizeDelta = new Vector2((Width - 124f) * _prog, 26f);
            _pct.text = Mathf.FloorToInt(_prog * 100f) + "%";
        }

        private void FinishBooking()
        {
            Finished = true;
            _status.text = "APPOINTMENT BOOKED.";
            Mgr.CompleteTicket(ShopAuditPopupManager.ApptRelief.Value);
            _closing = true;
            _closeT = 0.6f;
        }
